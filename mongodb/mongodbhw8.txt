db.sales.aggregate([

  { $match: { category: "Laptops" } }

])

db.sales.aggregate([

  { $group: { _id: "$product", totalQuantity: { $sum: "$quantity" } } }

])


db.sales.aggregate([

  {

    $group: {

      _id: "$product",

      averagePrice: { $avg: "$price" }

    }

  }

])


db.sales.aggregate([

  { $sort: { quantity: -1 } }

])
